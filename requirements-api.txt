# API 串接、資料處理與爬蟲用的套件。
#
# 與 requirements.txt（excel-to-db 本身）分開，但版本是相容的：
# 共用的 pandas / SQLAlchemy / pyodbc / openpyxl 釘在同一版，
# 三台機器裝完之後 excel-to-db 仍然跑得起來。
#
# 安裝：
#     pip install -r requirements-api.txt
#
# proxy 擋住（407）時走離線：在有網路的機器跑
#     fetch_wheels.bat 3.14 requirements-api.txt
# 再把 wheels\ 帶過去執行 install_offline.bat。
#
# 已驗證：Python 3.14 / win_amd64，每一個都有 cp314 wheel，不需要編譯器。


# --- 共用：與 requirements.txt 同版，不要單獨升級 ----------------------
# pandas 3.0 已經發布，不釘版本的話 pip 會裝 3.0.x，
# 那一版有 breaking change 且尚未驗證，excel-to-db 會壞。
pandas==2.3.3

xlrd==2.0.2


# --- API 串接 ----------------------------------------------------------

uvicorn[standard]==0.53.0       # 執行 FastAPI 服務
pydantic==2.13.5                # 資料驗證與格式轉換
pydantic-settings==2.15.0       # 系統設定與環境變數管理
python-multipart==0.0.32        # 檔案上傳與表單資料
python-dotenv==1.2.3            # 讀取 .env 設定檔
